cmake_minimum_required(VERSION 3.16)
project(educational_db_c C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(rb_tree rb_tree.c)
target_include_directories(rb_tree PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
enable_testing()
add_executable(test_rb_tree test_rb_tree.c)
add_test(NAME red_black_tree COMMAND test_rb_tree)

add_library(btree btree.c)
target_include_directories(btree PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(test_btree test_btree.c)
add_test(NAME btree COMMAND test_btree)

add_library(tree23 tree23.c)
target_link_libraries(tree23 PUBLIC rb_tree)
target_include_directories(tree23 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(test_tree23 test_tree23.c)
target_link_libraries(test_tree23 PRIVATE tree23)
add_test(NAME tree23 COMMAND test_tree23)

add_library(bplus bplus.c)
target_include_directories(bplus PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(test_bplus test_bplus.c)
add_test(NAME bplus COMMAND test_bplus)
